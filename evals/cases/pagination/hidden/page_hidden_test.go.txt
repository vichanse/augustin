package pagination

import (
	"reflect"
	"testing"
)

func TestHiddenPages(t *testing.T) {
	items := []string{"a", "b", "c", "d", "e"}
	for _, c := range []struct {
		page, size int
		want       []string
	}{
		{2, 2, []string{"c", "d"}}, {3, 2, []string{"e"}}, {4, 2, nil}, {0, 2, nil}, {1, 10, items}, {1, 0, nil},
	} {
		if got := Page(items, c.page, c.size); !reflect.DeepEqual(got, c.want) {
			t.Errorf("Page(%d, %d) = %v, want %v", c.page, c.size, got, c.want)
		}
	}
}
